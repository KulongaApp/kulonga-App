-- 1. TABELA ESCOLAS
CREATE TABLE IF NOT EXISTS escolas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nome TEXT NOT NULL,
  provincia TEXT NOT NULL,
  municipio TEXT,
  endereco TEXT,
  telefone TEXT,
  email TEXT,
  direccao_nome TEXT,
  direccao_telefone TEXT,
  criada_em TIMESTAMPTZ DEFAULT now(),
  actualizada_em TIMESTAMPTZ DEFAULT now()
);

-- 2. TABELA PROFESSORES
CREATE TABLE IF NOT EXISTS professores (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  escola_id UUID NOT NULL REFERENCES escolas(id) ON DELETE CASCADE,
  nome TEXT NOT NULL,
  email TEXT NOT NULL,
  telefone TEXT,
  disciplinas TEXT[],  -- array de disciplinas que lecciona
  activo BOOLEAN DEFAULT true,
  criado_em TIMESTAMPTZ DEFAULT now()
);

-- 3. TABELA ALUNOS
CREATE TABLE IF NOT EXISTS alunos (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  escola_id UUID NOT NULL REFERENCES escolas(id) ON DELETE CASCADE,
  nome_completo TEXT NOT NULL,
  data_nascimento DATE,
  genero TEXT CHECK (genero IN ('M', 'F')),
  foto_url TEXT,
  telefone TEXT,
  endereco TEXT,
  criado_em TIMESTAMPTZ DEFAULT now()
);

-- 4. TABELA TURMAS
CREATE TABLE IF NOT EXISTS turmas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  escola_id UUID NOT NULL REFERENCES escolas(id) ON DELETE CASCADE,
  nome TEXT NOT NULL,           -- ex: "8ªB", "10ªA"
  ano_lectivo TEXT NOT NULL,    -- ex: "2025/2026"
  serie TEXT,                   -- ex: "8ª classe"
  turno TEXT,                   -- ex: "Manhã", "Tarde"
  coordenador_id UUID REFERENCES professores(id),
  criada_em TIMESTAMPTZ DEFAULT now()
);

-- 5. TABELA DISCIPLINAS
CREATE TABLE IF NOT EXISTS disciplinas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  escola_id UUID NOT NULL REFERENCES escolas(id) ON DELETE CASCADE,
  nome TEXT NOT NULL,           -- ex: "Matemática"
  codigo TEXT,                  -- ex: "MAT"
  criada_em TIMESTAMPTZ DEFAULT now()
);

-- 6. RELAÇÃO TURMA-ALUNOS (matrícula)
CREATE TABLE IF NOT EXISTS turma_alunos (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  turma_id UUID NOT NULL REFERENCES turmas(id) ON DELETE CASCADE,
  aluno_id UUID NOT NULL REFERENCES alunos(id) ON DELETE CASCADE,
  matriculado_em TIMESTAMPTZ DEFAULT now(),
  UNIQUE(turma_id, aluno_id)
);

-- 7. RELAÇÃO TURMA-DISCIPLINAS
CREATE TABLE IF NOT EXISTS turma_disciplinas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  turma_id UUID NOT NULL REFERENCES turmas(id) ON DELETE CASCADE,
  disciplina_id UUID NOT NULL REFERENCES disciplinas(id) ON DELETE CASCADE,
  UNIQUE(turma_id, disciplina_id)
);

-- 8. RELAÇÃO DISCIPLINA-PROFESSOR
CREATE TABLE IF NOT EXISTS disciplina_professor (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  disciplina_id UUID NOT NULL REFERENCES disciplinas(id) ON DELETE CASCADE,
  professor_id UUID NOT NULL REFERENCES professores(id) ON DELETE CASCADE,
  turma_id UUID NOT NULL REFERENCES turmas(id) ON DELETE CASCADE,
  UNIQUE(disciplina_id, professor_id, turma_id)
);

-- 9. TABELA NOTAS
CREATE TABLE IF NOT EXISTS notas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  aluno_id UUID NOT NULL REFERENCES alunos(id) ON DELETE CASCADE,
  disciplina_id UUID NOT NULL REFERENCES disciplinas(id),
  turma_id UUID NOT NULL REFERENCES turmas(id),
  trimestre SMALLINT NOT NULL CHECK (trimestre IN (1, 2, 3)),
  tipo TEXT NOT NULL CHECK (tipo IN ('frequencia', 'prova', 'exame')),
  valor NUMERIC(4,2) NOT NULL CHECK (valor >= 0 AND valor <= 20),
  autor_id UUID NOT NULL,       -- user_id de quem lançou
  autor_papel TEXT NOT NULL CHECK (autor_papel IN ('professor', 'coordenador', 'secretaria')),
  lancado_em TIMESTAMPTZ DEFAULT now(),
  actualizado_em TIMESTAMPTZ DEFAULT now()
);

-- 10. TABELA ENCARRGADOS
CREATE TABLE IF NOT EXISTS encarregados (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  nome TEXT NOT NULL,
  telefone TEXT,
  criado_em TIMESTAMPTZ DEFAULT now()
);

-- 11. RELAÇÃO ENCARRGADO-ALUNO
CREATE TABLE IF NOT EXISTS encarregado_aluno (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  encarregado_id UUID NOT NULL REFERENCES encarregados(id) ON DELETE CASCADE,
  aluno_id UUID NOT NULL REFERENCES alunos(id) ON DELETE CASCADE,
  parentesco TEXT,              -- ex: "Pai", "Mãe", "Avó"
  UNIQUE(encarregado_id, aluno_id)
);

-- 12. TABELA TOKENS DE ACESSO
CREATE TABLE IF NOT EXISTS tokens_acesso (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  aluno_id UUID NOT NULL REFERENCES alunos(id) ON DELETE CASCADE,
  codigo TEXT NOT NULL UNIQUE,  -- 6 dígitos
  qr_hash TEXT,
  ativo BOOLEAN DEFAULT true,
  expira_em TIMESTAMPTZ,
  gerado_por UUID,             -- user_id da secretaria
  criado_em TIMESTAMPTZ DEFAULT now()
);

-- 13. TABELA NOTAS PENDENTES (sync offline)
CREATE TABLE IF NOT EXISTS notas_pendentes (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  aluno_id UUID NOT NULL,
  disciplina_id UUID NOT NULL,
  turma_id UUID NOT NULL,
  trimestre SMALLINT NOT NULL,
  tipo TEXT NOT NULL,
  valor NUMERIC(4,2) NOT NULL,
  autor_id UUID NOT NULL,
  autor_papel TEXT NOT NULL,
  status TEXT DEFAULT 'pendente' CHECK (status IN ('pendente', 'sync', 'erro')),
  lancado_em TIMESTAMPTZ DEFAULT now(),
  sincronizado_em TIMESTAMPTZ
);

-- VIEW: Notas vigentes (aplica hierarquia de papéis)
-- secretaria(3) > coordenador(2) > professor(1)
-- Em empate: nota mais recente vence
CREATE OR REPLACE VIEW notas_vigentes AS
WITH notas_com_rank AS (
  SELECT
    n.*,
    d.nome AS disciplina_nome,
    p.nome AS professor_nome,
    p.telefone AS professor_telefone,
    ROW_NUMBER() OVER (
      PARTITION BY n.aluno_id, n.disciplina_id, n.trimestre, n.tipo
      ORDER BY
        CASE n.autor_papel
          WHEN 'secretaria' THEN 3
          WHEN 'coordenador' THEN 2
          WHEN 'professor' THEN 1
        END DESC,
        n.actualizado_em DESC
    ) AS rank
  FROM notas n
  JOIN disciplinas d ON d.id = n.disciplina_id
  LEFT JOIN professores p ON p.id = n.autor_id
)
SELECT * FROM notas_com_rank WHERE rank = 1;

-- POLÍTICAS: Secretaria vê tudo da escola
CREATE POLICY "secretaria_ver_tudo" ON escolas
  FOR ALL USING (auth.uid() IN (
    SELECT user_id FROM professores WHERE escola_id = escolas.id
    AND 'secretaria' = ANY(
      SELECT jsonb_array_elements_text(
        (SELECT user_metadata FROM auth.users WHERE id = auth.uid())
      )
    )
  ));

-- POLÍTICAS: Professor vê as suas turmas e notas
CREATE POLICY "professor_ver_turmas" ON turmas
  FOR SELECT USING (id IN (
    SELECT turma_id FROM disciplina_professor WHERE professor_id IN (
      SELECT id FROM professores WHERE user_id = auth.uid()
    )
  ));

CREATE POLICY "professor_ver_notas" ON notas
  FOR SELECT USING (turma_id IN (
    SELECT turma_id FROM disciplina_professor WHERE professor_id IN (
      SELECT id FROM professores WHERE user_id = auth.uid()
    )
  ));

CREATE POLICY "professor_lancar_notas" ON notas
  FOR INSERT WITH CHECK (autor_id = auth.uid());

-- POLÍTICAS: Encarregado vê apenas o seu aluno
CREATE POLICY "encarregado_ver_aluno" ON notas
  FOR SELECT USING (aluno_id IN (
    SELECT aluno_id FROM encarregado_aluno WHERE encarregado_id IN (
      SELECT id FROM encarregados WHERE user_id = auth.uid()
    )
  ));

CREATE POLICY "encarregado_ver_aluno_info" ON alunos
  FOR SELECT USING (id IN (
    SELECT aluno_id FROM encarregado_aluno WHERE encarregado_id IN (
      SELECT id FROM encarregados WHERE user_id = auth.uid()
    )
  ));

-- POLÍTICAS: Tokens — qualquer um pode validar (lectura)
CREATE POLICY "validar_token" ON tokens_acesso
  FOR SELECT USING (ativo = true);

-- POLÍTICAS: Secretaria pode gerar tokens
CREATE POLICY "secretaria_gerar_token" ON tokens_acesso
  FOR INSERT WITH CHECK (gerado_por = auth.uid());

-- POLÍTICAS: Notas pendentes — cada utilizador vê as suas
CREATE POLICY "ver_notas_pendentes" ON notas_pendentes
  FOR SELECT USING (autor_id = auth.uid());

CREATE POLICY "criar_notas_pendentes" ON notas_pendentes
  FOR INSERT WITH CHECK (autor_id = auth.uid());

CREATE POLICY "actualizar_notas_pendentes" ON notas_pendentes
  FOR UPDATE USING (autor_id = auth.uid());

  FOR INSERT WITH CHECK (
    bucket_id = 'fotos-alunos'
    AND auth.role() = 'authenticated'
  );

  FOR SELECT USING (bucket_id = 'fotos-alunos');
